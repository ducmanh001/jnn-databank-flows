Câu truy vấn ClickHouse
SELECT a.answer_text AS age, g.answer_text AS gender, count() AS respondents
FROM respondent_answers_flat g
INNER JOIN respondent_answers_flat a USING (respondent_id, survey_id)
WHERE g.question_id = 'F1' AND a.question_id = 'F2' AND survey_id = 1
GROUP BY age, gender
ORDER BY age, gender
